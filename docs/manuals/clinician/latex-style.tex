% Retinal Review Workbench - shared print styling for the Quarto manuals.
% Goals: compact practical user-guide layout, no forced chapter pages,
% no hyphenation, no widows/orphans, restrained callouts.
\usepackage{xcolor}
\definecolor{rrwgreen}{HTML}{205C45}
\definecolor{rrwink}{HTML}{24312C}
\definecolor{rrwmuted}{HTML}{62716A}
\definecolor{rrwline}{HTML}{D5DFD9}
\definecolor{rrwtint}{HTML}{F2F6F3}

% Ragged-right text without hyphenation keeps long UI labels intact.
\usepackage[document]{ragged2e}
\hyphenpenalty=10000
\exhyphenpenalty=10000
\setlength{\RaggedRightParindent}{0pt}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.55em plus 0.15em}
% Body leading ~1.45x the font size (layout guidance: 1.5 for body text);
% code blocks and tables stay single-spaced.
\linespread{1.2}

% Widow and orphan control; allow the page bottom to vary instead of stretching.
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
\raggedbottom
\usepackage{needspace}

% Headings: chapters do not force a new page; headings never end a page.
\usepackage{titlesec}
\titleclass{\chapter}{straight}
\titleformat{\chapter}[hang]
  {\Needspace{10\baselineskip}\normalfont\color{rrwgreen}\fontsize{18}{22}\selectfont\bfseries}
  {\thechapter}{0.55em}{}
  [\vspace{3pt}{\color{rrwline}\titlerule[0.8pt]}]
\titlespacing*{\chapter}{0pt}{16pt}{10pt}
\titleformat{\section}
  {\Needspace{8\baselineskip}\normalfont\color{rrwgreen}\large\bfseries}
  {\thesection}{0.6em}{}
\titlespacing*{\section}{0pt}{14pt}{4pt}
\titleformat{\subsection}
  {\Needspace{5\baselineskip}\normalfont\color{rrwink}\normalsize\bfseries}
  {\thesubsection}{0.6em}{}
\titlespacing*{\subsection}{0pt}{10pt}{2pt}

% Header and footer.
\usepackage{fancyhdr}
\fancypagestyle{rrw}{%
  \fancyhf{}%
  \fancyhead[L]{\footnotesize\color{rrwmuted}Retinal Review Workbench \textbar{} \rrwDocTitle}%
  \fancyhead[R]{\footnotesize\color{rrwmuted}Software 0.7.0}%
  \fancyfoot[L]{\footnotesize\color{rrwmuted}\rrwDocId{} \textbar{} Version 0.7.0 \textbar{} 2026-09-23 \textbar{} \rrwFooterNote}%
  \fancyfoot[R]{\footnotesize\color{rrwgreen}\bfseries\thepage}%
  \renewcommand{\headrulewidth}{0.4pt}%
  \renewcommand{\headrule}{{\color{rrwline}\hrule height\headrulewidth width\headwidth}}%
  \renewcommand{\footrulewidth}{0pt}}
\pagestyle{rrw}
\AtBeginDocument{\fancypagestyle{plain}{\pagestyle{rrw}}}

% Lists.
\usepackage{enumitem}
\setlist{itemsep=0.15em,topsep=0.3em,parsep=0pt,leftmargin=1.6em}

% Figures and captions.
\usepackage{caption}
\captionsetup{font={small,color=rrwmuted},labelfont={color=rrwgreen,bf},skip=5pt,justification=raggedright,singlelinecheck=false}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}
\setlength{\fboxsep}{0pt}
\setlength{\fboxrule}{0.5pt}

% Tables.
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\renewcommand{\arraystretch}{1.15}
\AtBeginEnvironment{longtable}{\linespread{1}\selectfont}

% Table of contents: one compact page, then start the manual.
\makeatletter
\renewcommand{\contentsname}{Contents}
\let\rrw@oldtoc\tableofcontents
\renewcommand{\tableofcontents}{\begingroup\justifying\setlength{\parskip}{0.2em}\rrw@oldtoc\endgroup\clearpage}
\makeatother
